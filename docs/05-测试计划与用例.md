## 1. 测试策略

| 层次 | 对象 | 手段 | 频率 |
| --- | --- | --- | --- |
| 单元测试 | `physics.py`、`tilemap.py`、`entities/*`、`game.py`、`storage.py` 的纯逻辑 | pytest，无图形依赖 | 每次提交 |
| 集成测试 | 会话推进（移动+跳跃+踩踏+顶砖+死亡+重生）、存档往返 | pytest + `tmp_path` | 每次提交 |

| 打包冒烟 | exe 在干净机器 | 人工 | 每次打包 |

```python
import pytest
from src.physics import Body, move_and_collide, span
from src.tilemap import TileMap
from src.config import TILE


def make_map(rows):
    data = {"rows": len(rows), "cols": len(rows[0]), "tiles": rows, "theme": "grass"}
    return TileMap.from_dict(data)


def test_fall_lands_on_ground():
    tmap = make_map(["." * 10 for _ in range(17)] + ["#" * 10])
    body = Body(3 * TILE, 0, 26, 36, vy=780)
    for _ in range(200):
        contacts = move_and_collide(body, tmap, 1 / 120)
        if contacts.ground:
            break
    assert body.bottom == pytest.approx(17 * TILE)


def test_semisolid_only_from_above():
    rows = ["." * 10 for _ in range(10)] + ["." * 4 + "=" * 2 + "." * 4] + ["." * 10 for _ in range(7)]
    tmap = make_map(rows)
    falling = Body(4 * TILE + 7, 0, 26, 36, vy=700)
    for _ in range(120):
        contacts = move_and_collide(falling, tmap, 1 / 120)
        if contacts.ground:
            break
    assert falling.bottom == pytest.approx(10 * TILE)

    rising = Body(4 * TILE + 7, 13 * TILE, 26, 36, vy=-640)
    contacts = move_and_collide(rising, tmap, 1 / 120, drop_through=True)
    assert contacts.ground is False


def test_head_bump_reports_two_tiles():
    rows = ["." * 10 for _ in range(6)] + ["." * 3 + "BB" + "." * 5] + ["." * 10 for _ in range(11)]
    tmap = make_map(rows)
    body = Body(3 * TILE, 6 * TILE + 20, 26, 36, vy=-640)
    contacts = move_and_collide(body, tmap, 1 / 120)
    assert contacts.head is True
    assert sorted(contacts.head_tiles) == [(3, 6), (4, 6)]
```

`tests/test_player.py`：

```python
import pytest
from src.entities.player import Player
from src.entities.base import PowerState, WorldView, InputIntent
from src.physics import Body, move_and_collide
from src.tilemap import TileMap
from src.config import TILE, MAX_WALK, INVINCIBLE_MS


def flat_world():
    rows = ["." * 40 for _ in range(17)] + ["#" * 40]
    data = {"rows": 18, "cols": 40, "tiles": rows, "theme": "grass"}
    tmap = TileMap.from_dict(data)
    return tmap, WorldView(tmap, 0, 0, tmap.width_px)


def step(player, intent, tmap, world, steps=60):
    for _ in range(steps):
        player.update(1 / 120, intent, world)
        move_and_collide(player.body, tmap, 1 / 120)


def test_walk_speed_cap():
    tmap, world = flat_world()
    p = Player(2, 16)
    step(p, InputIntent(right=True), tmap, world, steps=30)
    assert p.body.vx == pytest.approx(MAX_WALK, abs=1.0)


def test_jump_height_full_hold():
    tmap, world = flat_world()
    p = Player(2, 16)
    p.update(1 / 120, InputIntent(jump_pressed=True, jump_held=True), world)
    start = p.body.y
    peak = start
    for _ in range(120):
        p.update(1 / 120, InputIntent(jump_held=True), world)
        move_and_collide(p.body, tmap, 1 / 120)
        peak = min(peak, p.body.y)
    assert 3.5 * TILE <= start - peak <= 4.6 * TILE


def test_big_hurt_shrinks_and_gets_invincible():
    p = Player(2, 16, power=PowerState.BIG)
    assert p.hurt() == "shrink"
    assert p.power is PowerState.SMALL
    assert p.invincible_ms == pytest.approx(INVINCIBLE_MS)
    assert p.hurt() == "none"
```

`tests/test_game.py`：

```python
import pytest
from src.game import LevelSession, SessionState
from src.entities.base import InputIntent
from src.storage import Profile


def make_session(tiles, spawn=(2, 12), entities=None, items=None):
    data = {
        "id": "test", "name": "测试", "theme": "grass",
        "cols": len(tiles[0]), "rows": len(tiles),
        "time": 300, "spawn": list(spawn), "tiles": tiles,
        "entities": entities or [], "items": items or {},
    }
    return LevelSession.from_dict(data, Profile())


def test_coin_collection():
    rows = ["." * 12 for _ in range(17)] + ["#" * 12]     # 只有最底一行是地面
    rows[16] = "..c........."                            # 金币在出生点右侧 1 格，同一行
    session = make_session(rows, spawn=(2, 16))
    for _ in range(60):
        session.step(InputIntent(right=True), 1 / 120)
        if session.coins:
            break
    assert session.coins == 1 and session.score == 100
    assert session.tilemap.tile_at(3, 16) == "."


def test_bump_brick_small_keeps_tile():
    rows = ["." * 10 for _ in range(18)]
    rows[17] = "#" * 10                                  # 地面
    rows[14] = "." * 4 + "B" + "." * 5                   # 砖块在头顶上方 2 格
    session = make_session(rows, spawn=(4, 16))
    for _ in range(480):
        session.step(InputIntent(jump_held=True, jump_pressed=True), 1 / 120)
        if any(e.kind == "bump" for e in session.events):
            break
    assert session.tilemap.tile_at(4, 14) == "B"        # SMALL 只弹不破


def test_death_and_respawn_resets_level():
    rows = ["." * 10 for _ in range(18)]
    session = make_session(rows)               # 无地形，直接落出关卡
    for _ in range(600):
        session.step(InputIntent(), 1 / 120)
        if session.state is SessionState.DYING:
            break
    for _ in range(200):
        session.step(InputIntent(), 1 / 120)
    assert session.lives == 2
    assert session.state is SessionState.RUNNING
    assert session.time_units == 300                      # 时间复位
    assert session.player.body.y == pytest.approx(13 * 40 - 36)   # 回到出生格
```

### 2.7 关卡校验用例

| 用例 | 场景 | 期望 |
| --- | --- | --- |
| UT-L-01 | 遍历 `assets/levels/*.json` | 每个文件均能 `TileMap.from_dict` 成功 |
| UT-L-02 | 每个关卡 `spawn` | 该格与上方一格均非实心 |
| UT-L-03 | 每个 `items` 键 | 指向的图块必须是 `?` |
| UT-L-04 | 每个 `entities` 的 `col/row` | 在 `[0, cols) × [0, rows)` 内 |
| UT-L-05 | 每个关卡均有且仅有一个 `goal` | 数量为 1，且 `goal.col` 距离右边界 ≤ 6 格 |
| UT-L-06 | 出生点周围 3 格 | 无敌人实体 |
| UT-L-07 | 可通关性脚本 | 从出生点以「最大跳高 4 格、保守水平跳距 5 格（设计约束）」做网格 BFS，能到达终点列 |

> UT-L-07 是防「设计出跳不过去的坑」的粗粒度静态检查：把实心/单向图块的可站立面当节点，按跳跃能力连边，BFS 到终点。它不替代人工试玩，但能在改关卡后立刻发现死路。

## 6. 缺陷管理

| 等级 | 定义 | 处理要求 |
| --- | --- | --- |
| P0 致命 | 无法启动、崩溃、丢存档、exe 在干净机器跑不起来、关卡不可加载 | 立即修复，阻断发布 |
| P1 严重 | 穿墙/卡地形、跳跃或踩踏手感明显错误、死亡不扣命、通关不解锁、音画严重不同步 | 发布前必须修复 |
| P2 一般 | 动画卡顿、相机轻微抖动、按钮点击偏移、文案错别字、音量步进异常 | 本版修复或明确延后 |
| P3 轻微 | 配色微调、粒子数量、特大幅度、非关键视觉细节 | 可延至下一版 |

缺陷记录字段：编号、等级、模块、复现步骤、实际结果、期望结果、环境（脚本/exe、是否全屏、是否 DEBUG）、附件截图或录屏、涉及的关卡与坐标。
